import{test}from'node:test';import assert from'node:assert/strict';
import{TrainingMedical}from'../apps/training/src/medical.js';import{MO2}from'../apps/training/src/medical-mo2-policy.js';import{registerBleedingWoundB1,updateBleedingB1}from'../apps/training/src/medical-b1.js';import{captureReferenceFlows}from'../apps/training/src/medical-b1-reference.js';import{REFERENCE_CIRCULATION,referenceFlow,freezeReferenceFlow}from'../apps/training/src/medical-reference-flow.js';import{registerLimbOpening,routeLimbVolume}from'../apps/training/src/medical-routing.js';import{bandageFraction,woundComplete}from'../apps/training/src/medical-hemostasis.js';import{pressureFixture,sampleMovement}from'../scripts/mo2b-fixtures.mjs';
function vascular(){const b=new TrainingMedical(),m=b.actor.medical;const add=id=>{const w={id,partKey:'leftLeg',s1OrganDamage:[{id:'left-femoral',kind:'vessel',vesselKind:'artery',domain:'leftLeg',bone:'leftUpperLeg',diameterM:.008,permanentVolumeM3:1e-8,direct:true,severity:.2,centerlineDistanceM:0,bulletRadiusM:.004}],bodyTrajectory:{entryAddress:{bone:'leftUpperLeg',localPoint:[.09,.8,0],localIncomingDirection:[0,0,-1]},woundChannel:{steps:[]}},bleedingProfile:{bleedingType:'none'}};m.wounds.push(w);registerBleedingWoundB1(m,w);registerLimbOpening(m,w,{entry:{},exit:{}});return w;};return{b,m,add};}
test('MO2b reference calculation reuses B1 at S1 reference circulation and leaves live clinical JSON byte-exact',()=>{const x=vascular(),w=x.add('a');x.m.hemodynamics.map=12;x.m.hemodynamics.cardiacOutputMul=.1;x.m.bloodMl=100;x.m.cardiacArrest=true;x.m.appliedTreatments=[{kind:'cat',status:'completed',target:{partKey:'leftLeg'}}];const before=JSON.stringify(x.m);captureReferenceFlows(x.b.actor);assert.equal(JSON.stringify(x.m),before);assert.deepEqual(REFERENCE_CIRCULATION.mapMmHg,93);assert.equal(REFERENCE_CIRCULATION.cardiacOutputMlMin,5250);const r=referenceFlow(w);assert.ok(Math.abs(r.total-87.5)<1e-10);assert.ok(Math.abs(r.external-r.total*.85)<1e-10);assert.ok(Math.abs(r.limbInternal-r.total*.15)<1e-10);assert.equal(r.cavity,0);assert.ok(Object.isFrozen(r));});
test('MO2b a later shared-vessel wound gets its new share while the original reference never changes',()=>{const x=vascular(),a=x.add('a');captureReferenceFlows(x.b.actor);const first=referenceFlow(a),b=x.add('b');captureReferenceFlows(x.b.actor);assert.strictEqual(referenceFlow(a),first);assert.ok(Math.abs(referenceFlow(b).total-first.total/2)<1e-10);updateBleedingB1(x.b.actor,0);assert.equal(x.m.bleedSources.length,1);assert.ok(Math.abs(x.m.bleedingModel.totalFlowMlPerS-87.5)<1e-10);assert.ok(first.total+referenceFlow(b).total>x.m.bleedingModel.totalFlowMlPerS);freezeReferenceFlow(a,{external:1});assert.strictEqual(referenceFlow(a),first);});
test('MO2b reference flow components distinguish a cavity source from verified skin and limb openings',()=>{const b=new TrainingMedical(),m=b.actor.medical,w={id:'cavity',partKey:'chest',b1Registered:true};m.wounds=[w];m.bleedSources=[{id:'vessel/aorta',destination:'thoracic_left',vesselKind:'artery',injuryType:'partial_laceration',woundIds:[w.id],normalFlowMlMin:5250,cumulativeMl:0,lesions:[{woundId:w.id,baseAreaMm2:20,currentAreaMm2:20,ageS:0}],flowMlPerS:0}];captureReferenceFlows(b.actor);const r=referenceFlow(w);assert.equal(r.external,0);assert.equal(r.limbInternal,0);assert.ok(r.cavity>0);assert.equal(r.total,r.cavity);});
test('MO2b fixed grades and bandage ratios survive30→12→45 circulation and a temporary zero-flow state',()=>{for(const ref of[30,18]){const x=pressureFixture(ref);x.w.care.bandage={active:true,finishing:false,appliedAtS:0};const grade=ref>20?'과다출혈':'일반 출혈',ratio=Math.min(1,20/ref);for(const q of[30,12,45,0,30]){const f=x.flow(12,q);assert.equal(f.grade,grade);assert.ok(Math.abs(f.actual-q*(1-ratio))<1e-10);assert.equal(woundComplete(x.m,x.w),ref<=20);}assert.equal(bandageFraction(x.m,x.w,0),ratio*.6);}});
test('MO2b unwrapping100/40/20 reference wounds restores exactly half the fixed blocked fraction at any pressure',()=>{for(const ref of[100,40,20]){const x=pressureFixture(ref),f=Math.min(1,20/ref);x.w.care.unwrappedFraction=f*.5;for(const current of[ref,12,45])assert.ok(Math.abs(x.flow(20,current).actual-current*(1-.5*f))<1e-10);}});
test('MO2b durability and efficiency are exactly180/250/350 at1 while all fixed times and work coefficients survive',()=>{for(const[k,max,seconds]of[['injector',180,5],['gauze',250,6],['foam',350,7]]){assert.equal(MO2.products[k].maximum,max);assert.equal(MO2.products[k].efficiency,1);assert.equal(MO2.products[k].seconds,seconds);}assert.equal(MO2.workPerAreaMm2,3);assert.equal(MO2.minimumComponentWork,2);assert.equal(MO2.movement.reopenFraction,.45);assert.equal(MO2.movement.seepSeconds,8);assert.equal(MO2.movement.seepFlowMlS,.3);});
test('MO2b walking and running experience targets are reproduced for all500 seeds with exclusive per-event counting',()=>{for(const[running,seep,reopen]of[[false,.45,.07],[true,.55,.60]]){const a=sampleMovement(running),b=sampleMovement(running);assert.deepEqual(a,b);assert.equal(a.seepTrialRate,seep);assert.equal(a.reopenTrialRate,reopen);for(const events of a.samples){assert.ok(events.filter(e=>e.kind==='reopen').length<=1);const times=events.map(e=>e.time);assert.equal(new Set(times).size,times.length);}}});

test('MO2b a new reference retains verified opening and limb-compartment state at birth',()=>{const x=vascular(),w=x.add('a');x.m.bleedSources[0].flowMlPerS=100;routeLimbVolume(x.m,x.m.bleedSources[0],100000);const before=JSON.stringify(x.m);captureReferenceFlows(x.b.actor);assert.equal(JSON.stringify(x.m),before);assert.equal(referenceFlow(w).limbInternal,0);assert.equal(referenceFlow(w).external,referenceFlow(w).total);});
