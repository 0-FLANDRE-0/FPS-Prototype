function setup(mode='realistic'){const inventory=new Inventory({includeMedical:true,includeAbdominal:true,includeMedic:true,treatmentMode:mode}),bridge=new TrainingMedical(),care=new SelfCare({inventory,getMedical:()=>bridge});for(const id of['ifak-01','medic-pouch-01','medic-support-01'])inventory.setMedicalContainerOpen(id);return{inventory,bridge,care,m:bridge.actor.medical};}
function finish(a,id){assert.ok(a.care.request('medical:'+id,{partKey:'abdomen'}),a.care.lastError);a.care.step(1000);return a.care.list.at(-1);}
function ticks(a,seconds){const start=a.bridge.time;for(let n=1;n<=Math.round(seconds*60);n++){a.care.step(1/60);a.bridge.tick(1/60,start+n/60);}}
test('M4b distinct3x3 medic pouch contains six physical supplies, mode cannot multiply them',()=>{for(const mode of['game','realistic','reality']){const a=setup(mode);assert.deepEqual(a.inventory.layout('container:medic-support-01'),{cols:3,rows:3});const items=[...a.inventory.items.values()].filter(x=>x.owner==='container:medic-support-01');assert.equal(items.length,6);assert.deepEqual(items.map(x=>x.definitionId),['pill','otfc','otfc','ketamine','ketamine','blood']);assert.ok(items.every(i=>i.remaining===1));assert.deepEqual(a.inventory.ownershipErrors(),[]);assert.equal(a.inventory.move('ammo:556_m855','container:medic-support-01',0,2),false);}});
test('M4b partial OTFC stops input immediately, re-bite preserves absorbed fraction and only one mouth',()=>{const a=setup('game'),t=finish(a,'otfc');ticks(a,15);assert.ok(Math.abs(t.absorbed-.25)<1e-10);assert.equal(a.care.request('medical:otfc-2',{partKey:'abdomen'}),false);assert.ok(a.care.stopOTFC());const absorbed=t.absorbed;ticks(a,10);assert.equal(t.absorbed,absorbed);assert.ok(a.care.reconnect(t.id));a.care.step(10);ticks(a,10);assert.ok(t.absorbed>absorbed);assert.equal(a.inventory.items.get('medical:otfc').remaining,1);assert.equal(a.inventory.items.get('medical:otfc').remainingFraction,1-t.absorbed);});
test('M4b OTFC automatic drop retains remainder, finished stick never returns on mode switch',()=>{const a=setup('game'),t=finish(a,'otfc');ticks(a,5);a.m.consciousness=60;a.bridge.tick(1/60,a.bridge.time+1/60);assert.equal(t.inMouth,false);assert.match(t.stopReason,/자동/);const left=t.absorbed;a.m.consciousness=100;a.m.isUnconscious=false;assert.ok(a.care.reconnect(t.id));a.care.step(10);ticks(a,80);assert.equal(t.absorbed,1);assert.equal(a.inventory.items.has('medical:otfc'),false);syncMedicalUses(a.inventory,'reality');syncMedicalUses(a.inventory,'game');assert.equal(a.inventory.items.has('medical:otfc'),false);assert.ok(left<1);});
